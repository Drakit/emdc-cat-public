<script>
(function () {
  var frame = document.getElementById("emdc-ld01-frame");
  var origin = "https://drakit.github.io";
  if (!frame) return;
  function init() {
    frame.contentWindow.postMessage({type:"EMDC_LD01_INIT"}, origin);
  }
  window.addEventListener("message", function (e) {
    if (e.origin !== origin || e.source !== frame.contentWindow || !e.data) return;
    if (e.data.type === "EMDC_LD01_HEIGHT") {
      var h = Number(e.data.height);
      if (Number.isFinite(h) && h >= 200 && h <= 100000) {
        frame.style.height = Math.ceil(h) + "px";
      }
    } else if (e.data.type === "EMDC_LD01_SCROLL") {
      var y = Number(e.data.top);
      if (Number.isFinite(y) && y >= 0 && y <= frame.clientHeight) {
        window.scrollTo({top:window.scrollY + frame.getBoundingClientRect().top + y - 20,behavior:"smooth"});
      }
    }
  });
  frame.addEventListener("load", init);
  init();
})();
</script>
